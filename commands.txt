================================================================================
                      ODOO 19 CLI COMMANDS CHEAT SHEET
================================================================================

--------------------------------------------------------------------------------
1. VIRTUAL ENVIRONMENT & DEPENDENCIES
--------------------------------------------------------------------------------

# Create Python virtual environment
python -m venv .venv

# Activate virtual environment
# Windows (PowerShell):
.\.venv\Scripts\Activate.ps1
# Windows (Command Prompt):
.\.venv\Scripts\activate.bat
# Linux / macOS / Git Bash:
source .venv/bin/activate

# Upgrade pip and packaging tools
python -m pip install --upgrade pip setuptools wheel

# Install dependencies for Community & Enterprise
pip install -r odoo-c/requirements.txt
pip install -r odoo-e/requirements.txt

# (Optional) Install PostgreSQL binary connector if needed
pip install psycopg2-binary


--------------------------------------------------------------------------------
2. RUNNING THE ODOO SERVER
--------------------------------------------------------------------------------

# Start Odoo using central configuration file (default)
python odoo-c/odoo-bin -c odoo.conf

# Start Odoo pointing directly to a specific database
python odoo-c/odoo-bin -c odoo.conf -d odoo-19

# Start with developer mode enabled (auto-reload code, live xml/qweb reload)
python odoo-c/odoo-bin -c odoo.conf --dev=reload,qweb,xml

# Start on a custom HTTP port (e.g. 8070)
python odoo-c/odoo-bin -c odoo.conf -p 8070

# Run from within the odoo-c directory:
# cd odoo-c
# python odoo-bin -c odoo.conf


--------------------------------------------------------------------------------
3. MODULE INSTALLATION & UPDATES (CLI)
--------------------------------------------------------------------------------

# Install one or more modules and stop when finished
python odoo-c/odoo-bin -c odoo.conf -d odoo-19 -i sale,crm --stop-after-init

# Update a specific custom or standard module and stop
python odoo-c/odoo-bin -c odoo.conf -d odoo-19 -u my_custom_addon --stop-after-init

# Update multiple modules
python odoo-c/odoo-bin -c odoo.conf -d odoo-19 -u sale,crm,account_accountant --stop-after-init

# Update ALL installed modules (database-wide refresh)
python odoo-c/odoo-bin -c odoo.conf -d odoo-19 -u all --stop-after-init

# Update module and keep the server running (convenient for testing)
python odoo-c/odoo-bin -c odoo.conf -d odoo-19 -u my_custom_addon


--------------------------------------------------------------------------------
4. MODULE SCAFFOLDING (CREATE NEW ADDON)
--------------------------------------------------------------------------------

# Scaffold a new custom addon module structure
python odoo-c/odoo-bin scaffold my_custom_addon custom_addons/


--------------------------------------------------------------------------------
5. INTERACTIVE ODOO SHELL (PYTHON REPL)
--------------------------------------------------------------------------------

# Launch interactive Python shell with full Odoo ORM environment initialized
python odoo-c/odoo-bin shell -c odoo.conf -d odoo-19

# Inside shell examples:
# >>> env['res.users'].search([])
# >>> env['res.partner'].search([('is_company', '=', True)])
# >>> env.cr.commit()


--------------------------------------------------------------------------------
6. RUNNING UNIT TESTS
--------------------------------------------------------------------------------

# Run unit tests for a specific module
python odoo-c/odoo-bin -c odoo.conf -d odoo-19 --test-enable --test-tags=/my_custom_addon --stop-after-init

# Run tests for a specific test class
python odoo-c/odoo-bin -c odoo.conf -d odoo-19 --test-enable --test-tags=:TestClassName --stop-after-init

# Run all standard tests at installation
python odoo-c/odoo-bin -c odoo.conf -d odoo-19 -i my_custom_addon --test-enable --stop-after-init


--------------------------------------------------------------------------------
7. POSTGRESQL DATABASE BACKUP & RESTORE (CLI)
--------------------------------------------------------------------------------

# Dump database to custom compressed format (.dump)
pg_dump -U odoo -h localhost -p 5432 -F c -b -v -f "backups/odoo-19_backup.dump" odoo-19

# Dump database to plain SQL script (.sql)
pg_dump -U odoo -h localhost -p 5432 -F p -v -f "backups/odoo-19_backup.sql" odoo-19

# Restore database from custom compressed format (.dump)
pg_restore -U odoo -h localhost -p 5432 -d odoo-19 -v "backups/odoo-19_backup.dump"

# Restore database from plain SQL format (.sql)
psql -U odoo -h localhost -p 5432 -d odoo-19 -f "backups/odoo-19_backup.sql"
